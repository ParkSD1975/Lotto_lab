    // 2. 각 필터 항목별 Min/Max 계산

    btn.classList.add('from-purple-500', 'to-purple-700');
    btnText.textContent = '필터 해제';

    }

    console.log('최근 10회차 필터 적용 완료');
}

## 3️⃣ **파일별 맞춤 함수 예시**

### 🔹 **합성수 분석 (composite_number.html)**
```javascript
function applyRecent10Filters() {
    const btn = document.getElementById('recent10FilterBtn');
    const btnText = document.getElementById('recent10FilterText');

    if (btn.classList.contains('from-purple-500')) {
        btn.classList.remove('from-purple-500', 'to-purple-700');
        btn.classList.add('from-blue-500', 'to-indigo-600');
        btnText.textContent = '최근 10회차 필터 적용';

        // 합성수 개수 필터 초기화 (0~6)
        for (let i = 0; i <= 6; i++) {
            const minInput = document.getElementById(`min_${i}`);
            const maxInput = document.getElementById(`max_${i}`);
            if (minInput) minInput.value = 0;
            if (maxInput) maxInput.value = 6;
        }

        highlightStartRound = null;
        renderList();
        return;
    }

    if (!allDrawData || allDrawData.length < 10) return;

    const recent10 = allDrawData.slice(0, 10);

    // 각 합성수 개수별(0~6) Min/Max 계산
    const counts = {};
    for (let i = 0; i <= 6; i++) {
        counts[i] = [];
    }

    recent10.forEach(draw => {
        const compositeCount = draw.numbers.filter(n => composites.has(n)).length;
        for (let i = 0; i <= 6; i++) {
            counts[i].push(compositeCount === i ? 1 : 0);
        }
    });

    // 필터 입력창에 값 적용
    for (let i = 0; i <= 6; i++) {
        const hasOccurrence = counts[i].some(v => v === 1);
        const minInput = document.getElementById(`min_${i}`);
        const maxInput = document.getElementById(`max_${i}`);
        
        if (minInput) minInput.value = hasOccurrence ? 1 : 0;
        if (maxInput) maxInput.value = hasOccurrence ? 1 : 0;
    }

    btn.classList.remove('from-blue-500', 'to-indigo-600');
    btn.classList.add('from-purple-500', 'to-purple-700');
    btnText.textContent = '필터 해제';

    document.getElementById('filterEnabled').checked = true;
    renderList();
    highlightRecent10(recent10[0].round);

    console.log('최근 10회차 합성수 필터 적용 완료');
}
```

### 🔹 **연속번호 분석 (consecutive_number.html)**
```javascript
function applyRecent10Filters() {
    const btn = document.getElementById('recent10FilterBtn');
    const btnText = document.getElementById('recent10FilterText');

    if (btn.classList.contains('from-purple-500')) {
        btn.classList.remove('from-purple-500', 'to-purple-700');
        btn.classList.add('from-blue-500', 'to-indigo-600');
        btnText.textContent = '최근 10회차 필터 적용';

        // 연속번호 개수 필터 초기화
        document.getElementById('minConsecutive').value = 0;
        document.getElementById('maxConsecutive').value = 6;

        highlightStartRound = null;
        renderList();
        return;
    }

    if (!allDrawData || allDrawData.length < 10) return;

    const recent10 = allDrawData.slice(0, 10);

    // 연속번호 개수 Min/Max 계산
    const consecutiveCounts = recent10.map(draw => {
        const sorted = [...draw.numbers].sort((a, b) => a - b);
        let count = 0;
        for (let i = 0; i < sorted.length - 1; i++) {
            if (sorted[i + 1] - sorted[i] === 1) count++;
        }
        return count;
    });

    const minVal = Math.min(...consecutiveCounts);
    const maxVal = Math.max(...consecutiveCounts);

    document.getElementById('minConsecutive').value = minVal;
    document.getElementById('maxConsecutive').value = maxVal;

    btn.classList.remove('from-blue-500', 'to-indigo-600');
    btn.classList.add('from-purple-500', 'to-purple-700');
    btnText.textContent = '필터 해제';

    document.getElementById('filterEnabled').checked = true;
    renderList();
    highlightRecent10(recent10[0].round);

    console.log('최근 10회차 연속번호 필터 적용 완료');
}
```

각 파일에 적용 시 다음을 확인하세요:

### ✅ **HTML 체크리스트**
- [ ] 조합 필터 섹션을 찾았는가?
- [ ] 버튼을 적절한 위치에 삽입했는가?

- [ ] 함수명이 `applyRecent10Filters`인가?
- [ ] 토글 로직이 올바른가? (purple ↔ blue)
- [ ] 필터 초기화 로직이 해당 파일에 맞는가?
- [ ] Min/Max 계산 로직이 해당 파일의 필터 구조에 맞는가?
- [ ] `renderList()` 함수를 호출하는가?
- [ ] 하이라이트 로직이 있는가?

- [ ] 버튼 클릭 시 색상이 변경되는가?
